% ====== Part 1: Union-Find ======
\section*{并查集 (Union-Find / Disjoint Set)}

\textbf{问题：} $N$ 个元素，支持 Find(查集合) 和 Union(合集合)。\\
\textbf{应用：} 等价关系、Kruskal MST、连通分量。

\subsection*{数据结构}

\begin{lstlisting}
int S[MAX];  // S[root] = -size (负数表示大小)
             // S[x] = parent (正数表示父节点索引)
\end{lstlisting}

\subsection*{初始化：每个元素自成一个集合}

\begin{lstlisting}
void init(int N) { for(int i=0;i<N;i++) S[i]=-1; }
\end{lstlisting}

\begin{multicols}{2}

\subsection*{Find — 路径压缩}

\begin{lstlisting}
int find(int x) {
    if (S[x] < 0) return x;        // 是根
    return S[x] = find(S[x]);      // 爹变祖宗！
}
\end{lstlisting}

\textbf{效果：} 查找路径上所有节点直连根，后续近乎 $O(1)$。

\subsection*{Union — 按大小合并}

\begin{lstlisting}
void un(int a, int b) {
    a=find(a); b=find(b);
    if(a==b) return;
    if(S[a]>S[b]){int t=a;a=b;b=t;} // a更大
    S[a]+=S[b];                     // 更新大小
    S[b]=a;                         // b挂到a
}
\end{lstlisting}

\textbf{树高：} $\le \lfloor\log_2 N\rfloor + 1$。

\columnbreak

\subsection*{四个版本对比}

\begin{tabular}{|c|c|c|}
\hline
\textbf{版本} & \textbf{操作} & \textbf{复杂度} \\
\hline
朴素 & 随便挂 & $\Theta(N^2)$ 最坏 \\
US & 小挂大 & $O(N+M\log N)$ \\
PC & 爹变祖宗 & 每操作均摊接近常数 \\
US+PC & 两者都有 & $O(M\alpha(M,N))$ \\
\hline
\end{tabular}

\begin{keybox}
$\alpha$ 是反 Ackermann 函数，$\alpha\le 4$ 在实际数据内。\\
\textbf{Union-by-Size + 路径压缩 = 几乎常数时间！}

\textbf{路径压缩不能和 Union-by-Height 一起用}（压缩改变高度），改用 Union-by-Rank。
\end{keybox}

\subsection*{$\log^*N$ (迭代对数)}

$\log^*65536=4$，$\log^*2^{65536}=5$。计算公式：反复取 $\log$ 到 $\le 1$ 的次数。

\subsection*{Ackermann 函数}

$A(0,j)=1$,$A(1,0)=2$,$A(i,0)=i+2(i\ge2)$,$A(i,j)=A(i-1,A(i,j-1))$（双递归，极快增长）。$\alpha$ 是其反函数——极慢增长。

\end{multicols}

% ====== Part 2: Graph ======
\section*{图论 (Graph)}

\subsection*{基本概念}

$G=(V,E)$。\textbf{无向图：} $(v_i,v_j)=(v_j,v_i)$。\textbf{有向图：} $\langle v_i,v_j\rangle \neq \langle v_j,v_i\rangle$。\\
\textbf{DAG：} 有向无环图。\textbf{握手定理：} $\sum deg(v)=2|E|$。

\subsection*{图的表示}

\begin{tabular}{|c|c|c|}
\hline
& \textbf{邻接矩阵} & \textbf{邻接表} \\
\hline
空间 & $O(|V|^2)$ & $O(|V|+|E|)$ \\
判相邻 & $O(1)$ & $O(deg)$ \\
适合 & 稠密图 & 稀疏图 \\
\hline
\end{tabular}

\noindent\textbf{有向图补逆邻接表/多重表}求入度。\\
\textbf{加权边：} 矩阵存 weight，表在节点加 weight 字段。

\begin{multicols}{2}

% ====== 拓扑排序 ======
\section*{1. 拓扑排序}

\textbf{目标：} 给 DAG 顶点线性排序，前驱在前。

\begin{lstlisting}
void topsort(Graph G) {
    Queue Q; int cnt=0;
    for each V if(Indegree[V]==0) Enqueue(V,Q);
    while(!IsEmpty(Q)){
        V=Dequeue(Q); TopNum[V]=++cnt;
        for each W adj to V
            if(--Indegree[W]==0) Enqueue(W,Q);
    }
    if(cnt<NumVertex) Error("有环!");
}
\end{lstlisting}

$T=O(|V|+|E|)$。朴素版每次扫描找入度0 $O(|V|^2)$。

\section*{2. AOV vs AOE}

\textbf{AOV：} 顶点=活动，边=先后关系。\\
\textbf{AOE：} 边=活动(有权)，顶点=事件。

\begin{itemize}
\item \textbf{EC[j]} = 事件 j 最早发生时间 = $\max_{i\to j}(EC[i]+C_{ij})$
\item \textbf{LC[i]} = 事件 i 最晚发生时间 = $\min_{i\to j}(LC[j]-C_{ij})$
\item \textbf{边 $\langle i,j\rangle$ 的 Slack} = LC[j] - EC[i] - C_{ij}，=0 在\textbf{关键路径}上
\end{itemize}

% ====== 最短路径 ======
\section*{3. 最短路径}

\subsection*{无权 — BFS}

BFS 就是无权图的最短路径。$O(|V|+|E|)$。\\
两个版本：邻接矩阵 $O(|V|^2)$，邻接表+队列 $O(|V|+|E|)$。

\subsection*{Dijkstra}

\textbf{贪心。} 每次选 Dist 最小的未处理点，松弛所有邻接。

\begin{lstlisting}
void dijkstra(int G[][MAX], int S, int N) {
    int known[MAX]={0}, dist[N];
    for(int i=0;i<N;i++) dist[i]=INF;
    dist[S]=0;
    for(;;){
        int V=-1, minD=INF;
        for(int i=0;i<N;i++)
            if(!known[i]&&dist[i]<minD)
                {minD=dist[i];V=i;}
        if(V==-1) break; known[V]=1;
        for(int W=0;W<N;W++)
            if(G[V][W] && !known[W])
                if(dist[V]+G[V][W]<dist[W])
                    dist[W]=dist[V]+G[V][W];
    }
}
\end{lstlisting}

$O(|V|^2)$ 朴素，$O(|E|\log|V|)$ 堆优化。\textbf{不能有负权边！}

\columnbreak

\subsection*{负权边 — Bellman-Ford}

每轮松弛所有边，共 $|V|-1$ 轮。$O(|V|\cdot|E|)$。\\
\textbf{第 |V| 轮还能松弛 $\to$ 有负权环！}

\subsection*{全源 — Floyd-Warshall}

\begin{lstlisting}
for(k=0;k<N;k++)for(i=0;i<N;i++)for(j=0;j<N;j++)
    if(D[i][k]+D[k][j]<D[i][j]) D[i][j]=D[i][k]+D[k][j];
\end{lstlisting}
$O(|V|^3)$。适合稠密图。

\subsection*{无权最短路径手算}

BFS 逐层扩展：初始 dist[S]=0,known[S]=T。每轮把所有刚 known 的邻接点 dist 更新为当前层+1。

\subsection*{Dijkstra 手算示例}

图有 7 个顶点 v1$\sim$v7，边权重见表。从 v1 开始。
\begin{enumerate}
\item v1 known：v2=2,v4=1
\item v4 known(dist=1,最小)：v3=1+2=3,v5=1+2=3,v6=1+8=9,v7=1+4=5
\item v2 known(dist=2)：v4已知跳过，v5=2+10=12>3 不变
\item v3 known(dist=3)：v6=3+5=8<9,更新v6=8
\item v5 known(dist=3)：v7=3+6=9>5,不变
\item v7 known(dist=5)
\item v6 known(dist=8)
\end{enumerate}
结果：v1(0),v2(2,v1),v3(3,v4),v4(1,v1),v5(3,v4),v6(8,v3),v7(5,v4)

% ====== MST ======
\section*{4. 最小生成树 (MST)}

\subsection*{Prim}

类似 Dijkstra，Dist = 到当前树的最短距离。\\
朴素 $O(|V|^2)$，适合稠密图。

\subsection*{Kruskal}

排序所有边 + Union-Find 判环。\\
$O(|E|\log|E|)$，适合稀疏图。

\begin{lstlisting}
void kruskal(Graph G){
    T={}; sort(edges by weight);
    while(T有<|V|-1条边){
        取最小边(u,v);
        if(find(u)!=find(v)){加入T; un(u,v);}
    }
}
\end{lstlisting}

\textbf{Prim vs Kruskal：} 稠密用Prim($O(|V|^2)$)，稀疏用Kruskal($O(|E|\log|E|)$)。

% ====== 网络流 ======
\section*{5. 网络流}

\textbf{核心技巧：残差图 + 反向边（撤销）。}

\begin{enumerate}
\item 残差图中找增广路
\item 流量 = 路径上最小残差容量
\item 正向减流量，反向加（等于增加残差）
\item 重复直到无增广路
\end{enumerate}

\textbf{三种变体：} BFS 找路 / 最大容量优先 / 最少边优先。\\
当 capmax 为小整数时，$O(|E|^2\log|V|)$。

\textbf{贪心不加反向边的问题：} s$\to$a$\to$d$\to$t 可能不是最优，需要反向边撤销。

% ====== DFS 应用 ======
\section*{6. DFS 应用}

\begin{lstlisting}
void DFS(Vertex V) {
    visited[V]=1;
    for(each W adj to V)
        if(!visited[W]) DFS(W);
}
\end{lstlisting}
$T=O(|V|+|E|)$。

\subsection*{双连通分量}

\textbf{Num[v] = DFS 编号}，\textbf{Low[v] = 通过回边能到的最小 Num}。

\begin{itemize}
\item 树边 (Tree edge)：父$\to$子
\item 回边 (Back edge)：后代$\to$祖先
\item 无向图 DFS 无前向边和交叉边
\end{itemize}

\textbf{关节点条件：}
\begin{itemize}
\item \textbf{v 是根} 且 $\ge 2$ 个孩子 $\to$ 关节点
\item \textbf{v 非根}：存在孩子 w，$Low[w] \ge Num[v]$ $\to$ v 是关节点
\end{itemize}

\textbf{计算 Low：}
\begin{lstlisting}
Low[v] = min{ Num[v],
              Num[w]  (if (v,w) is back edge),
              Low[w]  (if (v,w) is tree edge) }
\end{lstlisting}

\subsection*{强连通分量 (SCC)}

有向图。两次 DFS：原图 + 转置图。Tarjan 算法或 Kosaraju 算法。

\subsection*{欧拉回路}

\begin{itemize}
\item 无向图欧拉回路 $\iff$ 图连通 $\land$ 所有顶点度数为偶数
\item 无向图欧拉路径 $\iff$ 恰两个顶点度数为奇数（起点/终点）
\item 有向图欧拉回路 $\iff$ 每个点 in-degree = out-degree
\end{itemize}

\textbf{找回路：} DFS，每条边走完删除，回溯时记录顶点。用邻接表 + 删除边。

\subsection*{哈密顿回路}

经过每个顶点恰好一次的回路。NP 完全问题，没有多项式解。
